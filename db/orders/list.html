SQL used

SELECT o.order_id, c.customer_name, o.order_date, o.total_price,
               os.status_name
        FROM orders o
        LEFT JOIN customers c ON o.customer_id = c.customer_id
        LEFT JOIN order_statuses os ON o.status_cd = os.status_cd
        ORDER BY o.order_id
