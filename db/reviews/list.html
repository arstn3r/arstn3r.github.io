SQL used

SELECT r.review_id, c.customer_name, mi.item_name,
               r.star_rating, r.review_date
        FROM reviews r
        LEFT JOIN customers c ON r.customer_id = c.customer_id
        LEFT JOIN menu_items mi ON r.item_id = mi.item_id
        ORDER BY r.review_id
